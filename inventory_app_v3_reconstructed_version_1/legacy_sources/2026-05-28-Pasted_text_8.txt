def login_required(f):
    from functools import wraps
    @wraps(f)
    def decorated_function(*args, **kwargs):
        if "user_id" not in session:
            return redirect(url_for("login"))
        return f(*args, **kwargs)
    return decorated_function

def superadmin_required(f):
    from functools import wraps
    @wraps(f)
    def decorated_function(*args, **kwargs):
        if session.get("role") != "superadmin":
            return "Access Denied", 403
        return f(*args, **kwargs)
    return decorated_function

def to_float(value):
    if value is None:
        return 0.0
    if isinstance(value, Decimal):
        return float(value)
    try:
        return float(value)
    except (TypeError, ValueError):
        return 0.0

def get_unified_vms(run_id_vmware=None, run_id_hyperv=None, vcenter_list=None,
                    host_name=None, cluster_name=None, search_term=None,
                    page=1, per_page=100):
    conn = get_db_connection()
    cursor = conn.cursor()
    vmware_run_id, hyperv_run_id = get_latest_run_ids()
    run_id_vmware = run_id_vmware or vmware_run_id
    run_id_hyperv = run_id_hyperv or hyperv_run_id
    offset = (page - 1) * per_page

    vmware_where = "WHERE vm.run_id = %s"
    hyperv_where = "WHERE run_id = %s"
    params_vmware = [run_id_vmware]
    params_hyperv = [run_id_hyperv]

    if vcenter_list:
        placeholders = ','.join(['%s'] * len(vcenter_list))
        vmware_where += f" AND vc.vcenter_name IN ({placeholders})"
        params_vmware.extend(vcenter_list)
        hyperv_where += f" AND cluster_name IN ({placeholders})"
        params_hyperv.extend(vcenter_list)

    if host_name:
        vmware_where += " AND vm.hostname = %s"
        params_vmware.append(host_name)
        hyperv_where += " AND host_name = %s"
        params_hyperv.append(host_name)

    if cluster_name:
        hyperv_where += " AND cluster_name = %s"
        params_hyperv.append(cluster_name)

    if search_term:
        like = f"%{search_term}%"
        vmware_where += " AND (vm.vm_name LIKE %s OR vm.ip_address LIKE %s OR vm.os LIKE %s)"
        params_vmware.extend([like, like, like])
        hyperv_where += " AND (vm_name LIKE %s OR ip_address LIKE %s OR os LIKE %s)"
        params_hyperv.extend([like, like, like])

    vmware_query = f"""
        SELECT
            vm.vm_name, vm.cpu, vm.ram_mb, vm.os, vm.ip_address,
            vm.power_state, vm.hostname as host_name, NULL as cluster_name,
            'VMware' as platform, vc.vcenter_name as vcenter_name, vm.total_disk_gb
        FROM vm_inventory vm
        JOIN vcenters vc ON vm.vcenter_id = vc.vcenter_id
        {vmware_where}
    """

    hyperv_query = f"""
        SELECT
            vm_name, cpu, ram_mb, os, ip_address,
            power_state, host_name, cluster_name,
            'Hyper-V' as platform, cluster_name as vcenter_name, total_disk_gb
        FROM hyperv_vms
        {hyperv_where}
    """

    cursor.execute(f"SELECT COUNT(*) as total FROM ({vmware_query} UNION ALL {hyperv_query}) as combined",
                   params_vmware + params_hyperv)
    total = cursor.fetchone()["total"]

    full_query = f"""
        SELECT * FROM (
            {vmware_query}
            UNION ALL
            {hyperv_query}
        ) as combined
        ORDER BY platform, vm_name
        LIMIT %s OFFSET %s
    """
    all_params = params_vmware + params_hyperv + [per_page, offset]
    cursor.execute(full_query, all_params)
    results = cursor.fetchall()
    cursor.close()
    conn.close()
    return {"data": results, "total": total, "page": page, "per_page": per_page,
            "pages": (total + per_page - 1) // per_page}

            session["role"] = role

@app.route("/api/clusters")
@login_required
def get_clusters():
    vcenter = request.args.get("vcenter")
    run_id_vmware = request.args.get("run_id_vmware")
    run_id_hyperv = request.args.get("run_id_hyperv")
    conn = get_db_connection()
    cursor = conn.cursor()
    vmware_run_id, hyperv_run_id = get_latest_run_ids()
    run_id_vmware = run_id_vmware or vmware_run_id
    run_id_hyperv = run_id_hyperv or hyperv_run_id

    filter_names = None
    if vcenter and vcenter.strip() and vcenter.lower() != "all":
        filter_names = [v.strip() for v in vcenter.split(",") if v.strip()]

    queries = []
    params = []

    vmware_query = """
        SELECT
            c.cluster_name,
            vc.vcenter_name,
            dc.datacenter_name,
            c.num_hosts,
            c.num_vms,
            c.num_cpu_cores AS total_cores,
            c.num_cpu_threads,
            c.total_cpu_mhz,
            c.used_cpu_mhz,
            c.free_cpu_mhz,
            c.cpu_usage_percent,
            c.total_memory_mb,
            c.used_memory_mb,
            c.free_memory_mb,
            c.memory_usage_percent,
            c.total_storage_gb,
            c.used_storage_gb,
            c.free_storage_gb,
            c.storage_usage_percent,
            c.ha_enabled,
            c.drs_enabled,
            'VMware' as platform
        FROM clusters c
        JOIN vcenters vc ON c.vcenter_id = vc.vcenter_id
        LEFT JOIN datacenters dc ON c.vcenter_id = dc.vcenter_id AND dc.run_id = c.run_id
        WHERE c.run_id = %s
    """
    queries.append(vmware_query)
    params.append(run_id_vmware)
    if filter_names:
        placeholders = ','.join(['%s'] * len(filter_names))
        queries[-1] += f" AND vc.vcenter_name IN ({placeholders})"
        params.extend(filter_names)

    hyperv_query = """
        SELECT
            cluster_name,
            cluster_name AS vcenter_name,
            NULL AS datacenter_name,
            num_hosts,
            num_vms,
            total_cpu_cores AS total_cores,
            total_cpu_threads,
            total_cpu_mhz,
            used_cpu_mhz,
            free_cpu_mhz,
            cpu_usage_percent,
            total_memory_mb,
            used_memory_mb,
            free_memory_mb,
            memory_usage_percent,
            total_storage_gb,
            used_storage_gb,
            free_storage_gb,
            storage_usage_percent,
            NULL AS ha_enabled,
            NULL AS drs_enabled,
            'Hyper-V' as platform
        FROM hyperv_clusters
        WHERE run_id = %s
    """
    queries.append(hyperv_query)
    params.append(run_id_hyperv)
    if filter_names:
        placeholders = ','.join(['%s'] * len(filter_names))
        queries[-1] += f" AND cluster_name IN ({placeholders})"
        params.extend(filter_names)

    full_query = " UNION ALL ".join(queries) + " ORDER BY platform, cluster_name"
    cursor.execute(full_query, params)
    data = cursor.fetchall()
    cursor.close()
    conn.close()
    return jsonify(data)

@app.route("/api/datastores")
@login_required
def datastores():
    vcenter = request.args.get("vcenter")
    cluster = request.args.get("cluster")
    run_id_vmware = request.args.get("run_id_vmware")
    run_id_hyperv = request.args.get("run_id_hyperv")
    conn = get_db_connection()
    cursor = conn.cursor()
    vmware_run_id, hyperv_run_id = get_latest_run_ids()
    run_id_vmware = run_id_vmware or vmware_run_id
    run_id_hyperv = run_id_hyperv or hyperv_run_id

    filter_names = None
    if vcenter and vcenter.strip() and vcenter.lower() != "all":
        filter_names = [v.strip() for v in vcenter.split(",") if v.strip()]

    queries = []
    params = []

    # VMware datastores (convert MB to GB, ensure numeric)
    vmware_query = """
        SELECT
            d.datastore_name,
            d.status,
            d.is_accessible,
            d.type,
            d.num_vms,
            COALESCE(ROUND(d.capacity_mb / 1024, 2), 0) AS capacity_gb,
            COALESCE(ROUND(d.used_mb / 1024, 2), 0) AS used_gb,
            COALESCE(ROUND(d.free_mb / 1024, 2), 0) AS free_gb,
            COALESCE(ROUND(d.free_percent, 2), 0) AS free_percent,
            COALESCE(ROUND(d.provisioned_mb / 1024, 2), 0) AS provisioned_gb,            
            d.cluster_name,
            vc.vcenter_name,
            'VMware' as platform
        FROM datastores d
        JOIN vcenters vc ON d.vcenter_id = vc.vcenter_id
        WHERE d.run_id = %s
    """
    queries.append(vmware_query)
    params.append(run_id_vmware)
    if filter_names:
        placeholders = ','.join(['%s'] * len(filter_names))
        queries[-1] += f" AND vc.vcenter_name IN ({placeholders})"
        params.extend(filter_names)
    if cluster:
        queries[-1] += " AND d.cluster_name = %s"
        params.append(cluster)

    # Hyper-V datastores
    hyperv_query = """
        SELECT
            datastore_name,
            status,
            is_accessible,
            type,
            num_vms,
            COALESCE(capacity_gb, 0) AS capacity_gb,
            COALESCE(used_gb, 0) AS used_gb,
            COALESCE(free_gb, 0) AS free_gb,
            COALESCE(free_percent, 0) AS free_percent,
            NULL AS provisioned_gb,
            cluster_name,
            cluster_name AS vcenter_name,
            'Hyper-V' as platform
        FROM hyperv_datastores
        WHERE run_id = %s
    """
    queries.append(hyperv_query)
    params.append(run_id_hyperv)
    if filter_names:
        placeholders = ','.join(['%s'] * len(filter_names))
        queries[-1] += f" AND cluster_name IN ({placeholders})"
        params.extend(filter_names)
    if cluster:
        queries[-1] += " AND cluster_name = %s"
        params.append(cluster)

    full_query = " UNION ALL ".join(queries) + " ORDER BY platform, datastore_name"
    cursor.execute(full_query, params)
    data = cursor.fetchall()
    cursor.close()
    conn.close()
    return jsonify(data)

@app.route("/api/datastore_vms")
@login_required
def datastore_vms():
    datastore = request.args.get("datastore")
    cluster = request.args.get("cluster")
    run_id_vmware = request.args.get("run_id_vmware")
    run_id_hyperv = request.args.get("run_id_hyperv")
    conn = get_db_connection()
    cursor = conn.cursor()
    vmware_run_id, hyperv_run_id = get_latest_run_ids()
    if run_id_vmware and run_id_vmware not in (None, "", "undefined"):
        try:
            vmware_run_id = int(run_id_vmware)
        except ValueError:
            pass
    if run_id_hyperv and run_id_hyperv not in (None, "", "undefined"):
        try:
            hyperv_run_id = int(run_id_hyperv)
        except ValueError:
            pass

    results = []
    if not datastore:
        cursor.close()
        conn.close()
        return jsonify(results)

    datastore_clean = datastore.strip()

    # VMware
    if cluster and cluster not in (None, "", "undefined"):
        cursor.execute("""
            SELECT DISTINCT
                vm.vm_name, vm.cpu, vm.ram_mb, vm.os,
                vm.ip_address, vm.total_disk_gb, vc.vcenter_name,
                'VMware' as platform
            FROM vm_inventory vm
            JOIN vm_disks d ON vm.id = d.vm_id AND d.run_id = vm.run_id
            JOIN vcenters vc ON vm.vcenter_id = vc.vcenter_id
            JOIN datastores ds ON ds.run_id = vm.run_id
                AND TRIM(SUBSTRING_INDEX(SUBSTRING_INDEX(d.disk_path, ']', 1), '[', -1)) = ds.datastore_name
            WHERE vm.run_id = %s
              AND ds.datastore_name = %s
              AND ds.cluster_name = %s
        """, (vmware_run_id, datastore_clean, cluster))
        results.extend(cursor.fetchall())
    else:
        cursor.execute("""
            SELECT DISTINCT
                vm.vm_name, vm.cpu, vm.ram_mb, vm.os,
                vm.ip_address, vm.total_disk_gb, vc.vcenter_name,
                'VMware' as platform
            FROM vm_inventory vm
            JOIN vm_disks d ON vm.id = d.vm_id AND d.run_id = vm.run_id
            JOIN vcenters vc ON vm.vcenter_id = vc.vcenter_id
            WHERE vm.run_id = %s
              AND TRIM(SUBSTRING_INDEX(SUBSTRING_INDEX(d.disk_path, ']', 1), '[', -1)) = %s
        """, (vmware_run_id, datastore_clean))
        results.extend(cursor.fetchall())

    # Hyper-V
    if cluster and cluster not in (None, "", "undefined"):
        cursor.execute("SELECT hyperv_cluster_id FROM hyperv_clusters_master WHERE cluster_name = %s", (cluster,))
        cluster_row = cursor.fetchone()
        if cluster_row:
            cluster_id = cluster_row['hyperv_cluster_id']
            cursor.execute("""
                SELECT volume_path
                FROM hyperv_datastores
                WHERE run_id = %s AND hyperv_cluster_id = %s AND LOWER(datastore_name) = LOWER(%s)
            """, (hyperv_run_id, cluster_id, datastore_clean))
            ds_row = cursor.fetchone()
            if ds_row:
                volume_path = ds_row['volume_path']
                if not volume_path.endswith('\\'):
                    volume_path += '\\'
                pattern = volume_path.lower() + '%'
                cursor.execute("""
                    SELECT DISTINCT
                        v.vm_name, v.cpu, v.ram_mb, v.os,
                        v.ip_address, v.total_disk_gb, v.cluster_name as vcenter_name,
                        'Hyper-V' as platform
                    FROM hyperv_vms v
                    JOIN hyperv_vm_disks d ON v.id = d.vm_id AND d.run_id = v.run_id
                    WHERE v.run_id = %s
                      AND v.hyperv_cluster_id = %s
                      AND LOWER(d.disk_path) LIKE %s
                """, (hyperv_run_id, cluster_id, pattern))
                results.extend(cursor.fetchall())
    cursor.close()
    conn.close()
    return jsonify(results)

def _latest_run(cursor, vcenter_id):
    for table in ("clusters", "hosts", "datastores", "vm_inventory"):
        try:
            cursor.execute(f"SELECT MAX(run_id) AS r FROM `{table}` WHERE vcenter_id = %s", (vcenter_id,))
            row = cursor.fetchone()
            if row and row["r"] is not None:
                return row["r"]
        except Exception:
            continue
    return None

def _run_popen(cmd, log_list):
    proc = subprocess.Popen(cmd, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True)
    for line in iter(proc.stdout.readline, ""):
        log_list[0] += line
    proc.wait()
    return proc.returncode

        rc = _run_popen(cmd, log)
        job["vms"][idx]["log"] += log[0]
        if rc != 0:
            job["vms"][idx]["status"] = "failed"
            continue
        user_data = vm.get("user_data", "").strip()
        if not user_data:
            job["vms"][idx]["status"] = "success"
            continue
        job["vms"][idx]["log"] += "\n─── Post-Install Script ───────────────────────\n"
        log2 = [""]
        if vm_os == "windows":
            win_user = vm.get("vm_admin_user", "Administrator")
            win_pass = vm.get("vm_admin_password", "")
            ps_cmd = [
                "ansible", vm["vm_ip"],
                "-m", "win_shell",
                "-a", user_data,
                "-e", f"ansible_user={win_user}",
                "-e", f"ansible_password={win_pass}",
                "-e", "ansible_connection=winrm",
                "-e", "ansible_winrm_transport=ntlm",
                "-e", "ansible_winrm_server_cert_validation=ignore",
            ]
        else:
            ps_cmd = [
                "ansible", vm["vm_ip"],
                "-u", "root",
                "-m", "shell",
                "-a", user_data,
                "--ssh-extra-args", "-o StrictHostKeyChecking=no -o ConnectTimeout=30",
            ]
        rc2 = _run_popen(ps_cmd, log2)
        job["vms"][idx]["log"] += log2[0]
        if rc2 != 0:
            job["vms"][idx]["status"] = "failed"
            continue
        job["vms"][idx]["status"] = "success"
    job["status"] = "done" if all(v["status"] == "success" for v in job["vms"]) else "done_with_errors"

def train_cpu_forecast_model(cluster_name, historical_data):
    """
    historical_data: list of dicts with keys 'run_timestamp' and 'cpu_usage_percent'
    Assumes data is sorted by time (oldest first).
    Trains a model to predict next CPU usage based on last 3 readings.
    """
    if len(historical_data) < 4:
        return None

    # Create features: usage at t-3, t-2, t-1
    df = pd.DataFrame(historical_data)
    df = df.sort_values('run_timestamp')
    usages = df['cpu_usage_percent'].values

    X, y = [], []
    for i in range(3, len(usages)):
        X.append([usages[i-3], usages[i-2], usages[i-1]])
        y.append(usages[i])

    X = np.array(X)
    y = np.array(y)

    model = LinearRegression()
    model.fit(X, y)

    # Save model
    joblib.dump(model, os.path.join(MODEL_DIR, f'{cluster_name}_cpu_model.pkl'))
    return model

def predict_next_cpu_usage(cluster_name, last_three):
    """
    last_three: list of last 3 CPU usage percentages.
    Returns predicted next value.
    """
    model_path = os.path.join(MODEL_DIR, f'{cluster_name}_cpu_model.pkl')
    if not os.path.exists(model_path):
        return None
    model = joblib.load(model_path)
    X_pred = np.array([last_three]).reshape(1, -1)
    return model.predict(X_pred)[0]

# ---------- VM Placement Suggestions Engine ----------
def get_vm_placement_suggestions(vm_requirements, run_id_vmware=None, run_id_hyperv=None):
    conn = get_db_connection()
    cursor = conn.cursor()
    vmware_run_id, hyperv_run_id = get_latest_run_ids()
    run_id_vmware = run_id_vmware or vmware_run_id
    run_id_hyperv = run_id_hyperv or hyperv_run_id

    req_cpu_cores = vm_requirements.get('cpu_cores', 2)
    req_ram_gb = vm_requirements.get('ram_gb', 4)
    req_storage_gb = vm_requirements.get('storage_gb', 50)

    # Get average CPU MHz per core from VMware hosts (fallback 2500)
    cursor.execute("SELECT AVG(cpu_mhz) as avg_mhz FROM hosts WHERE run_id = %s", (vmware_run_id,))
    row = cursor.fetchone()
    avg_cpu_mhz = row['avg_mhz'] if row and row['avg_mhz'] else 2500

    req_cpu_mhz = req_cpu_cores * avg_cpu_mhz
    req_ram_mb = req_ram_gb * 1024

    suggestions = []

    # VMware clusters
    if vm_requirements.get('platform') in ['any', 'vmware']:
        cursor.execute("""
            SELECT
                c.cluster_name,
                vc.vcenter_name,
                c.num_hosts,
                c.num_vms,
                c.free_cpu_mhz,
                c.free_memory_mb,
                c.free_storage_gb,
                c.cpu_usage_percent,
                c.memory_usage_percent,
                c.storage_usage_percent,
                c.total_cpu_mhz,
                c.total_memory_mb,
                c.total_storage_gb,
                'VMware' as platform
            FROM clusters c
            JOIN vcenters vc ON c.vcenter_id = vc.vcenter_id
            WHERE c.run_id = %s
              AND c.free_cpu_mhz >= %s
              AND c.free_memory_mb >= %s
              AND c.free_storage_gb >= %s
        """, (run_id_vmware, req_cpu_mhz, req_ram_mb, req_storage_gb))

        for row in cursor.fetchall():
            cpu_remaining_ratio = (row['free_cpu_mhz'] - req_cpu_mhz) / (row['total_cpu_mhz'] or 1)
            mem_remaining_ratio = (row['free_memory_mb'] - req_ram_mb) / (row['total_memory_mb'] or 1)
            storage_remaining_ratio = (row['free_storage_gb'] - req_storage_gb) / (row['total_storage_gb'] or 1)

            cpu_score = min(100, max(0, cpu_remaining_ratio * 100))
            mem_score = min(100, max(0, mem_remaining_ratio * 100))
            storage_score = min(100, max(0, storage_remaining_ratio * 100))

            load_penalty = 1.0
            if row['cpu_usage_percent'] > 70:
                load_penalty *= 0.7
            if row['memory_usage_percent'] > 70:
                load_penalty *= 0.7
            if row['storage_usage_percent'] > 70:
                load_penalty *= 0.7

            total_score = (cpu_score * 0.4 + mem_score * 0.35 + storage_score * 0.25) * load_penalty

            suggestions.append({
                'platform': 'VMware',
                'cluster_name': row['cluster_name'],
                'vcenter_name': row['vcenter_name'],
                'resource_name': f"{row['cluster_name']} ({row['vcenter_name']})",
                'total_score': round(total_score, 2),
                'scores': {
                    'cpu': round(cpu_score, 2),
                    'ram': round(mem_score, 2),
                    'storage': round(storage_score, 2)
                },
                'available_resources': {
                    'free_cpu_mhz': row['free_cpu_mhz'],
                    'free_ram_gb': round(row['free_memory_mb'] / 1024, 2),
                    'free_storage_gb': round(row['free_storage_gb'], 2),
                    'current_cpu_usage': round(row['cpu_usage_percent'], 2),
                    'current_memory_usage': round(row['memory_usage_percent'], 2),
                    'current_storage_usage': round(row['storage_usage_percent'], 2)
                },
                'num_hosts': row['num_hosts'],
                'num_vms': row['num_vms']
            })

    # Hyper‑V clusters
    if vm_requirements.get('platform') in ['any', 'hyperv']:
        cursor.execute("""
            SELECT
                cluster_name,
                num_hosts,
                num_vms,
                free_cpu_mhz,
                free_memory_mb,
                free_storage_gb,
                cpu_usage_percent,
                memory_usage_percent,
                storage_usage_percent,
                total_cpu_mhz,
                total_memory_mb,
                total_storage_gb,
                'Hyper-V' as platform
            FROM hyperv_clusters
            WHERE run_id = %s
              AND free_cpu_mhz >= %s
              AND free_memory_mb >= %s
              AND free_storage_gb >= %s
        """, (run_id_hyperv, req_cpu_mhz, req_ram_mb, req_storage_gb))

        for row in cursor.fetchall():
            cpu_remaining_ratio = (row['free_cpu_mhz'] - req_cpu_mhz) / (row['total_cpu_mhz'] or 1)
            mem_remaining_ratio = (row['free_memory_mb'] - req_ram_mb) / (row['total_memory_mb'] or 1)
            storage_remaining_ratio = (row['free_storage_gb'] - req_storage_gb) / (row['total_storage_gb'] or 1)

            cpu_score = min(100, max(0, cpu_remaining_ratio * 100))
            mem_score = min(100, max(0, mem_remaining_ratio * 100))
            storage_score = min(100, max(0, storage_remaining_ratio * 100))

            load_penalty = 1.0
            if row['cpu_usage_percent'] and row['cpu_usage_percent'] > 70:
                load_penalty *= 0.7
            if row['memory_usage_percent'] and row['memory_usage_percent'] > 70:
                load_penalty *= 0.7
            if row['storage_usage_percent'] and row['storage_usage_percent'] > 70:
                load_penalty *= 0.7

            total_score = (cpu_score * 0.4 + mem_score * 0.35 + storage_score * 0.25) * load_penalty

            suggestions.append({
                'platform': 'Hyper-V',
                'cluster_name': row['cluster_name'],
                'resource_name': row['cluster_name'],
                'total_score': round(total_score, 2),
                'scores': {
                    'cpu': round(cpu_score, 2),
                    'ram': round(mem_score, 2),
                    'storage': round(storage_score, 2)
                },
                'available_resources': {
                    'free_cpu_mhz': row['free_cpu_mhz'],
                    'free_ram_gb': round(row['free_memory_mb'] / 1024, 2) if row['free_memory_mb'] else 0,
                    'free_storage_gb': round(row['free_storage_gb'], 2) if row['free_storage_gb'] else 0,
                    'current_cpu_usage': round(row['cpu_usage_percent'], 2) if row['cpu_usage_percent'] else 0,
                    'current_memory_usage': round(row['memory_usage_percent'], 2) if row['memory_usage_percent'] else 0,
                    'current_storage_usage': round(row['storage_usage_percent'], 2) if row['storage_usage_percent'] else 0
                },
                'num_hosts': row['num_hosts'],
                'num_vms': row['num_vms']
            })

    cursor.close()
    conn.close()
    suggestions.sort(key=lambda x: x['total_score'], reverse=True)
    return suggestions

def train_models():
    """Train forecasting models for all clusters"""
    conn = get_db_connection()
    cursor = conn.cursor()
    results = []

    # VMware clusters
    cursor.execute("SELECT DISTINCT cluster_name FROM clusters")
    clusters = cursor.fetchall()
    for cluster in clusters:
        cluster_name = cluster['cluster_name']
        cursor.execute("""
            SELECT r.run_timestamp, c.cpu_usage_percent
            FROM clusters c
            JOIN vc_runs r ON c.run_id = r.run_id
            WHERE c.cluster_name = %s
            ORDER BY r.run_timestamp ASC

            results.append({'cluster': cluster_name, 'status': 'trained' if model else 'failed'})
        else:
            results.append({'cluster': cluster_name, 'status': 'insufficient data'})

    # Hyper‑V clusters
    cursor.execute("SELECT DISTINCT cluster_name FROM hyperv_clusters")
    hyperv_clusters = cursor.fetchall()
    for cluster in hyperv_clusters:
        cluster_name = cluster['cluster_name']
        cursor.execute("""
            SELECT r.run_timestamp, h.cpu_usage_percent
            FROM hyperv_clusters h
            JOIN hyperv_runs r ON h.run_id = r.run_id
            WHERE h.cluster_name = %s
            ORDER BY r.run_timestamp ASC
            LIMIT 30
        """, (cluster_name,))
        hist = cursor.fetchall()
        if len(hist) >= 4:
            data = [{'run_timestamp': row['run_timestamp'], 'cpu_usage_percent': row['cpu_usage_percent']} for row in hist]
            train_cpu_forecast_model(cluster_name, data)
            results.append({'cluster': cluster_name, 'status': 'trained'})
        else:
            results.append({'cluster': cluster_name, 'status': 'insufficient data'})

    cursor.close()
    conn.close()

def get_aggregated_site_data(run_id, site_name, vcenter_list):
    conn = get_db_connection()
    cursor = conn.cursor()
    if not run_id:
        cursor.execute("SELECT MAX(run_id) as max_run FROM vc_runs")
        run_id = cursor.fetchone()["max_run"]
    placeholders = ','.join(['%s'] * len(vcenter_list))
    cursor.execute(f"""
        SELECT
            SUM(num_vms) as total_vms,
            SUM(total_cores) as total_cores,
            SUM(used_cpu_mhz) as used_cpu_mhz,
            SUM(free_cpu_mhz) as free_cpu_mhz,
            SUM(total_cpu_vms) as total_cpu_vms,
            SUM(total_memory_gb) as total_memory_gb,
            SUM(used_memory_gb) as used_memory_gb,
            SUM(free_memory_gb) as free_memory_gb,
            SUM(total_memory_vms_gb) as total_memory_vms_gb,
            SUM(total_storage_gb) as total_storage_gb,
            SUM(used_storage_gb) as used_storage_gb,
            SUM(free_storage_gb) as free_storage_gb,
            SUM(num_clusters) as total_clusters,
            SUM(num_hosts) as total_hosts,
            SUM(num_datastores) as total_datastores,
            AVG(cpu_usage_percent) as avg_cpu_usage_percent,
            AVG(memory_usage_percent) as avg_memory_usage_percent,
            AVG(storage_usage_percent) as avg_storage_usage_percent,
            GROUP_CONCAT(DISTINCT Vcenter) as vcenter_list
        FROM vcenter_summary_all_runs
        WHERE run_id = %s AND Vcenter IN ({placeholders})
    """, (run_id, *vcenter_list))
    aggregated_data = cursor.fetchone()
    cursor.execute("SELECT run_id, run_timestamp FROM vc_runs ORDER BY run_id DESC")
    runs = cursor.fetchall()
    cursor.close()
    conn.close()
    data = []
    if aggregated_data and aggregated_data['total_vms'] is not None:
        avg_cpu = aggregated_data['avg_cpu_usage_percent'] or 0
        avg_memory = aggregated_data['avg_memory_usage_percent'] or 0
        avg_storage = aggregated_data['avg_storage_usage_percent'] or 0
        total_cpu_vms = aggregated_data['total_cpu_vms'] or 0
        total_cores = aggregated_data['total_cores'] or 1
        subscription_cpu_ratio = round(total_cpu_vms / total_cores, 2) if total_cores > 0 else 0
        total_memory_vms_gb = aggregated_data['total_memory_vms_gb'] or 0
        total_memory_gb = aggregated_data['total_memory_gb'] or 1
        subscription_memory_ratio = round(total_memory_vms_gb / total_memory_gb, 2) if total_memory_gb > 0 else 0
        aggregated_row = {
            'Vcenter': f'{site_name}_Site_Aggregated',
            'ip_vcenter': '',
            'num_clusters': aggregated_data['total_clusters'] or 0,
            'num_hosts': aggregated_data['total_hosts'] or 0,
            'num_datastores': aggregated_data['total_datastores'] or 0,
            'num_vms': aggregated_data['total_vms'] or 0,
            'total_cores': aggregated_data['total_cores'] or 0,
            'cpu_usage_percent': round(avg_cpu, 2),
            'used_cores': aggregated_data['used_cpu_mhz'] or 0,
            'free_cores': aggregated_data['free_cpu_mhz'] or 0,
            'total_memory_gb': aggregated_data['total_memory_gb'] or 0,
            'memory_usage_percent': round(avg_memory, 2),
            'used_memory_gb': aggregated_data['used_memory_gb'] or 0,
            'free_memory_gb': aggregated_data['free_memory_gb'] or 0,
            'total_storage_gb': aggregated_data['total_storage_gb'] or 0,
            'storage_usage_percent': round(avg_storage, 2),
            'used_storage_gb': aggregated_data['used_storage_gb'] or 0,
            'free_storage_gb': aggregated_data['free_storage_gb'] or 0,
            'Subscription_cpu_ratio': subscription_cpu_ratio,
            'total_cpu_vms': aggregated_data['total_cpu_vms'] or 0,
            'Subscription_memory_ratio': subscription_memory_ratio,
            'total_memory_vms_gb': aggregated_data['total_memory_vms_gb'] or 0,
            'Subscription_storage_ratio': 0,
            'total_disk_size_gb': 0,
            'vcenter_list': aggregated_data['vcenter_list'] or ''
        }
        data.append(aggregated_row)
    return jsonify({"run_id": run_id, "data": data, "runs": runs})

# ── safe numeric conversion (fixes Decimal * float TypeError) ─

def _f(val, div=1, dec=1):
    """Convert any DB value (Decimal/int/float/None) → plain Python float."""
    try:
        return round(float(val) / float(div), dec) if val is not None else 0.0
    except (TypeError, ValueError):
        return 0.0

def _favg(lst, key):
    vals = [float(x[key]) for x in lst if x.get(key) is not None]
    return round(sum(vals) / len(vals), 1) if vals else 0.0

def _kw(text, *kws):
    return any(k in text for k in kws)

# ── LLM caller ─────────────────────────────────────────────────

def _llm(prompt, model="phi3-local:latest"):
    """Try phi3 variants, then tinyllama. Always returns a string."""
    for m in (model, "phi3:latest", "phi3-mini:latest", "tinyllama:latest"):
        try:
            r = requests.post(
                "http://localhost:11434/api/generate",
                json={"model": m, "prompt": prompt, "stream": False,
                      "options": {"temperature": 0.1, "num_ctx": 3072, "num_predict": 350}},
                timeout=90,
            )
            if r.status_code == 200:
                raw = r.json().get("response", "").strip()
                for mk in ("### Answer", "Answer:", "### Question", "Question:"):
                    if mk in raw:
                        raw = raw.split(mk)[-1].strip()
                if raw:
                    return raw
        except Exception:
            continue
    return None   # caller handles None

def ai_train_models():
    conn = get_db_connection(); cursor = conn.cursor()
    results = []

    def _train(key, rows):
        hist = [{"run_timestamp": r["run_timestamp"],
                 "cpu_usage_percent": float(r["pct"] or 0)}
                for r in rows if r.get("pct") is not None]
        if len(hist) >= 4:
            m = train_cpu_forecast_model(key, hist)
            return "trained" if m else "failed"
        return "insufficient data"

    # VMware clusters
    cursor.execute("SELECT DISTINCT cluster_name FROM clusters")
    for row in cursor.fetchall():
        cn = row["cluster_name"]
        cursor.execute("""
            SELECT r.run_timestamp, c.cpu_usage_percent AS pct
            FROM clusters c JOIN vc_runs r ON c.run_id=r.run_id
            WHERE c.cluster_name=%s ORDER BY r.run_timestamp ASC LIMIT 60
        """, (cn,))
        results.append({"entity": f"[VMware Cluster] {cn}",
                        "status": _train(cn, cursor.fetchall())})

    # Hyper-V clusters
    cursor.execute("SELECT DISTINCT cluster_name FROM hyperv_clusters")
    for row in cursor.fetchall():
        cn = row["cluster_name"]
        cursor.execute("""
            SELECT r.run_timestamp, h.cpu_usage_percent AS pct
            FROM hyperv_clusters h JOIN hyperv_runs r ON h.run_id=r.run_id
            WHERE h.cluster_name=%s ORDER BY r.run_timestamp ASC LIMIT 60
        """, (cn,))
        results.append({"entity": f"[Hyper-V Cluster] {cn}",
                        "status": _train(cn, cursor.fetchall())})

    # VMware Hosts (memory % as proxy)
    cursor.execute("""
        SELECT DISTINCT host_name FROM hosts
        WHERE run_id=(SELECT MAX(run_id) FROM vc_runs)
    """)
    for row in cursor.fetchall():
        hn = row["host_name"]
        cursor.execute("""
            SELECT r.run_timestamp,
                   CASE WHEN memory_total_gb>0
                        THEN ROUND(memory_used_gb/memory_total_gb*100,2)
                        ELSE 0 END AS pct
            FROM hosts h JOIN vc_runs r ON h.run_id=r.run_id
            WHERE h.host_name=%s ORDER BY r.run_timestamp ASC LIMIT 60
        """, (hn,))
        results.append({"entity": f"[Host] {hn}",
                        "status": _train("host__"+hn, cursor.fetchall())})

    # VMware Datastores (used % trend)
    cursor.execute("""
        SELECT DISTINCT datastore_name FROM datastores
        WHERE run_id=(SELECT MAX(run_id) FROM vc_runs)
    """)
    for row in cursor.fetchall():
        dn = row["datastore_name"]
        cursor.execute("""
            SELECT r.run_timestamp,
                   CASE WHEN capacity_mb>0
                        THEN ROUND(used_mb/capacity_mb*100,2)
                        ELSE 0 END AS pct
            FROM datastores d JOIN vc_runs r ON d.run_id=r.run_id
            WHERE d.datastore_name=%s ORDER BY r.run_timestamp ASC LIMIT 60
        """, (dn,))
        results.append({"entity": f"[Datastore] {dn}",
                        "status": _train("ds__"+dn, cursor.fetchall())})

    cursor.close(); conn.close()
    trained = sum(1 for r in results if r["status"] == "trained")
    return jsonify({"results": results, "trained": trained, "total": len(results)})

        def hist_ds(months, label):

        def trend_report(metric, n_runs=5):
            conn2 = get_db_connection(); c2 = conn2.cursor()
            c2.execute("""
                SELECT run_id, run_timestamp FROM vc_runs
                ORDER BY run_id DESC LIMIT %s
            """, (n_runs,))
            runs = list(reversed(c2.fetchall()))
            col_map = {"cpu":"cpu_usage_percent","memory":"memory_usage_percent",
                       "ram":"memory_usage_percent","storage":"storage_usage_percent"}
            db_col = col_map.get(metric, "cpu_usage_percent")
            pivot = {}
            for run in runs:
                c2.execute(f"""
                    SELECT c.cluster_name, c.{db_col} AS val, vc.vcenter_name
                    FROM clusters c JOIN vcenters vc ON c.vcenter_id=vc.vcenter_id
                    WHERE c.run_id=%s
                """, (run["run_id"],))
                for r in c2.fetchall():
                    cn = r["cluster_name"]
                    if cn not in pivot:
                        pivot[cn] = {"name":cn, "vcenter":r["vcenter_name"]}
                    ts = str(run["run_timestamp"])[:16]
                    pivot[cn][ts] = round(float(r["val"] or 0), 1)
            c2.close(); conn2.close()
            ts_labels = [str(r["run_timestamp"])[:16] for r in runs]
            cols = ([{"key":"name","label":"Cluster"},{"key":"vcenter","label":"vCenter"}]
                  + [{"key":ts,"label":ts} for ts in ts_labels])
            return _tbl(f"{metric.upper()} Trend — last {n_runs} runs",
                        cols, list(pivot.values()))

        def safe_place(reqs):
            try:
                raw = get_vm_placement_suggestions(reqs)
                clean = []

        def parse_reqs(q):
            cpu=2; ram=4; stor=50; plat="any"
            m = re.search(r'(\d+)\s*(?:vcpu|vcore|cpu|core)', q)
            if m: cpu = int(m.group(1))

            m = re.search(r'(\d+)\s*gb\s*(?:ram|memory|mem)', q)
            if m: ram = int(m.group(1))

            m = re.search(r'(\d+)\s*gb\s*(?:disk|storage|stor)', q)
            if m: stor = int(m.group(1))
            if "vmware" in q: plat="vmware"
            elif _kw(q,"hyper-v","hyperv"): plat="hyperv"
            return {"cpu_cores":cpu,"ram_gb":ram,"storage_gb":stor,"platform":plat}

        def direct(q):

            if _kw(q,"full report","infrastructure report","complete report"):

                return None, hist_ds(1,"1 month")
            if _kw(q,"datastore last 3","new datastore last 3"):
                return None, hist_ds(3,"3 months")

            m_t = re.search(r'(?:last|for)\\s*(\\d+)\\s*run', q)

                return None, trend_report("cpu", n_r)

                frag = em.group(1).strip()

                            return None, det

                if mh:

            if _kw(q,"where should i place","where to place","where can i place",
                     "suggest placement","best cluster for","best environment",

                     "placement for","best place for"):

                     "number of vm","vm count","count vm","all vm",

                    pred=f"\n  Predicted next: {ml_pred[c['name']]}%" if c["name"] in ml_pred else ""

                top=sorted(all_clusters,key=lambda x:-x["free_ram_gb"])[:5]
                lines=["Clusters with most free RAM:"]

            if _kw(q,"most free storage","free storage","cluster with most storage"):

@app.route("/api/reporting/overview")
@login_required
def reporting_overview():
    """Get overview statistics including storage and both platforms (all clusters)"""
    conn = get_db_connection()
    cursor = conn.cursor()
    vmware_run_id, hyperv_run_id = get_latest_run_ids()

    # ----- VMware -----
    cursor.execute("""
        SELECT COUNT(*) as count,
               AVG(cpu_usage_percent) as avg_cpu,
               AVG(memory_usage_percent) as avg_mem,
               AVG(storage_usage_percent) as avg_storage
        FROM clusters WHERE run_id = %s
    """, (vmware_run_id,))
    vmw_cluster = cursor.fetchone()

    cursor.execute("SELECT COUNT(*) as count FROM hosts WHERE run_id = %s", (vmware_run_id,))
    vmw_hosts = cursor.fetchone()
    cursor.execute("SELECT COUNT(*) as count FROM vm_inventory WHERE run_id = %s", (vmware_run_id,))
    vmw_vms = cursor.fetchone()
    cursor.execute("SELECT COUNT(*) as count FROM datastores WHERE run_id = %s", (vmware_run_id,))
    vmw_ds = cursor.fetchone()

    # ----- Hyper‑V -----
    cursor.execute("""
        SELECT COUNT(*) as count,
               AVG(cpu_usage_percent) as avg_cpu,
               AVG(memory_usage_percent) as avg_mem,
               AVG(storage_usage_percent) as avg_storage
        FROM hyperv_clusters WHERE run_id = %s
    """, (hyperv_run_id,))
    hv_cluster = cursor.fetchone()

    cursor.execute("SELECT COUNT(*) as count FROM hyperv_hosts WHERE run_id = %s", (hyperv_run_id,))
    hv_hosts = cursor.fetchone()
    cursor.execute("SELECT COUNT(*) as count FROM hyperv_vms WHERE run_id = %s", (hyperv_run_id,))
    hv_vms = cursor.fetchone()
    cursor.execute("SELECT COUNT(*) as count FROM hyperv_datastores WHERE run_id = %s", (hyperv_run_id,))
    hv_ds = cursor.fetchone()

    # ----- ALL clusters (no LIMIT) for the chart -----
    cursor.execute("""
        SELECT cluster_name, 
               COALESCE(cpu_usage_percent, 0) as cpu_usage_percent,
               COALESCE(memory_usage_percent, 0) as memory_usage_percent,
               COALESCE(storage_usage_percent, 0) as storage_usage_percent,
               'VMware' as platform
        FROM clusters WHERE run_id = %s
        UNION ALL
        SELECT cluster_name,
               COALESCE(cpu_usage_percent, 0),
               COALESCE(memory_usage_percent, 0),
               COALESCE(storage_usage_percent, 0),
               'Hyper-V' as platform
        FROM hyperv_clusters WHERE run_id = %s
        ORDER BY platform, cluster_name
    """, (vmware_run_id, hyperv_run_id))
    all_clusters = cursor.fetchall()

    cursor.close()
    conn.close()

    total_clusters = (vmw_cluster['count'] or 0) + (hv_cluster['count'] or 0)
    total_hosts    = (vmw_hosts['count'] or 0) + (hv_hosts['count'] or 0)
    total_vms      = (vmw_vms['count'] or 0) + (hv_vms['count'] or 0)
    total_ds       = (vmw_ds['count'] or 0) + (hv_ds['count'] or 0)

    def safe_avg(vmw_val, hv_val, total):
        if total == 0:
            return 0
        return round(((vmw_val or 0) + (hv_val or 0)) / 2, 1)

    avg_cpu = safe_avg(vmw_cluster['avg_cpu'], hv_cluster['avg_cpu'], total_clusters)
    avg_mem = safe_avg(vmw_cluster['avg_mem'], hv_cluster['avg_mem'], total_clusters)
    avg_storage = safe_avg(vmw_cluster['avg_storage'], hv_cluster['avg_storage'], total_clusters)

    # Prepare chart data (use cluster name + platform for clarity)
    labels = [f"{c['cluster_name']} ({c['platform']})" for c in all_clusters]
    cpu_data = [float(c['cpu_usage_percent']) for c in all_clusters]
    mem_data = [float(c['memory_usage_percent']) for c in all_clusters]
    storage_data = [float(c['storage_usage_percent']) for c in all_clusters]

    return jsonify({
        "summary": {
            "total_clusters": total_clusters,
            "total_hosts": total_hosts,
            "total_vms": total_vms,
            "total_datastores": total_ds,
            "avg_cpu_usage": avg_cpu,
            "avg_memory_usage": avg_mem,
            "avg_storage_usage": avg_storage,
            "vmware_clusters": vmw_cluster['count'] or 0,
            "hyperv_clusters": hv_cluster['count'] or 0,
            "vmware_hosts": vmw_hosts['count'] or 0,
            "hyperv_hosts": hv_hosts['count'] or 0,
            "vmware_vms": vmw_vms['count'] or 0,
            "hyperv_vms": hv_vms['count'] or 0
        },
        "chart": {
            "labels": labels,
            "cpu_data": cpu_data,
            "memory_data": mem_data,
            "storage_data": storage_data
        }
    })

def reporting_clusters():
    """Get clusters report with filters"""
    platform = request.args.get('platform')
    min_cpu = request.args.get('min_cpu')
    max_cpu = request.args.get('max_cpu')

    conn = get_db_connection()
    cursor = conn.cursor()
    vmware_run_id, hyperv_run_id = get_latest_run_ids()

    results = []

    # VMware clusters
    if not platform or platform == 'VMware' or platform == 'all':
        query = """
            SELECT c.*, vc.vcenter_name, 'VMware' as platform
            FROM clusters c
            JOIN vcenters vc ON c.vcenter_id = vc.vcenter_id
            WHERE c.run_id = %s
        """
        params = [vmware_run_id]

        if min_cpu:
            query += " AND c.cpu_usage_percent >= %s"
            params.append(float(min_cpu))
        if max_cpu:
            query += " AND c.cpu_usage_percent <= %s"
            params.append(float(max_cpu))

        cursor.execute(query, params)
        results.extend(cursor.fetchall())

    # Hyper-V clusters
    if not platform or platform == 'Hyper-V' or platform == 'all':
        query = "SELECT *, cluster_name as vcenter_name, 'Hyper-V' as platform FROM hyperv_clusters WHERE run_id = %s"
        params = [hyperv_run_id]

        if min_cpu:
            query += " AND cpu_usage_percent >= %s"
            params.append(float(min_cpu))
        if max_cpu:
            query += " AND cpu_usage_percent <= %s"
            params.append(float(max_cpu))

        cursor.execute(query, params)
        results.extend(cursor.fetchall())

    cursor.close()
    conn.close()

    # Convert Decimal to float for JSON serialization
    for row in results:
        for key, value in row.items():
            if isinstance(value, Decimal):
                row[key] = float(value)

    return jsonify(results)

    free_threshold = request.args.get('free_threshold')

    conn = get_db_connection()
    cursor = conn.cursor()
    vmware_run_id, hyperv_run_id = get_latest_run_ids()

    results = []

    # VMware datastores
    if not platform or platform == 'VMware' or platform == 'all':
        query = """

            JOIN vcenters vc ON d.vcenter_id = vc.vcenter_id
            WHERE d.run_id = %s
        """
        params = [vmware_run_id]

        if free_threshold:
            query += " AND d.free_percent < %s"

        cursor.execute(query, params)
        results.extend(cursor.fetchall())

    # Hyper-V datastores
    if not platform or platform == 'Hyper-V' or platform == 'all':

    conn.close()

    for row in results:
        for key, value in row.items():
            if isinstance(value, Decimal):
                row[key] = float(value)

    return jsonify(results)

@app.route("/api/reporting/trends")
@login_required
def reporting_trends():
    """Get trend analysis data for charts (fixed for MariaDB)"""
    metric = request.args.get('metric', 'cpu')
    runs = int(request.args.get('runs', 10))
    
    conn = get_db_connection()
    cursor = conn.cursor()
    
    # Step 1: Get the top 5 clusters by CPU usage from the latest run
    cursor.execute("""
        SELECT cluster_name
        FROM clusters
        WHERE run_id = (SELECT MAX(run_id) FROM vc_runs)
        ORDER BY cpu_usage_percent DESC
        LIMIT 5
    """)
    top_clusters = [row['cluster_name'] for row in cursor.fetchall()]
    
    if not top_clusters:
        return jsonify({'labels': [], 'datasets': []})
    
    # Step 2: Fetch historical data for those clusters
    placeholders = ','.join(['%s'] * len(top_clusters))
    query = f"""
        SELECT c.cluster_name, r.run_timestamp,
               c.cpu_usage_percent, c.memory_usage_percent, c.storage_usage_percent
        FROM clusters c
        JOIN vc_runs r ON c.run_id = r.run_id
        WHERE c.cluster_name IN ({placeholders})
        ORDER BY c.cluster_name, r.run_timestamp ASC
    """
    cursor.execute(query, top_clusters)
    rows = cursor.fetchall()
    cursor.close()
    conn.close()
    
    # Organize data by cluster
    clusters_data = {}
    for row in rows:
        cluster = row['cluster_name']
        if cluster not in clusters_data:
            clusters_data[cluster] = {'timestamps': [], 'values': []}
        # Keep only last 'runs' entries per cluster
        clusters_data[cluster]['timestamps'].append(str(row['run_timestamp'])[:10])
        if metric == 'cpu':
            val = float(row['cpu_usage_percent'] or 0)
        elif metric == 'memory':
            val = float(row['memory_usage_percent'] or 0)
        else:
            val = float(row['storage_usage_percent'] or 0)
        clusters_data[cluster]['values'].append(val)
    
    # Trim to last 'runs' entries and prepare chart data
    datasets = []
    colors = ['#667eea', '#764ba2', '#f093fb', '#4facfe', '#43e97b']
    for i, (cluster, data) in enumerate(clusters_data.items()):
        # Keep only the most recent 'runs' values
        if len(data['values']) > runs:
            data['values'] = data['values'][-runs:]
            data['timestamps'] = data['timestamps'][-runs:]
        datasets.append({
            'label': cluster,
            'data': data['values'],
            'borderColor': colors[i % len(colors)],
            'backgroundColor': colors[i % len(colors)] + '20',
            'fill': True,
            'tension': 0.4
        })
    
    # Use timestamps from the first cluster (all should have same length)
    timestamps = list(clusters_data.values())[0]['timestamps'] if clusters_data else []
    
    return jsonify({
        'labels': timestamps,
        'datasets': datasets
    })






from datetime import datetime, timedelta

@app.route("/api/reporting/historical")
@login_required
def reporting_historical():
    """Get historical data for VMs, Hosts, or Datastores.
       Period can be: 1w,2w,3w,1m,3m,6m,9m,1y
    """
    report_type = request.args.get('type', 'vms')
    period = request.args.get('period', '1m')

    period_days = {
        '1w': 7, '2w': 14, '3w': 21,
        '1m': 30, '3m': 90, '6m': 180, '9m': 270, '1y': 365
    }.get(period, 30)

    cutoff_date = (datetime.now() - timedelta(days=period_days)).strftime('%Y-%m-%d %H:%M:%S')

    conn = get_db_connection()
    cursor = conn.cursor()
    results = []

    if report_type == 'vms':
        cursor.execute("""
            SELECT vm.vm_name, 'VMware' as platform, vc.vcenter_name,
                   vm.cpu, vm.ram_mb, vm.os, r.run_timestamp as created_at, r.run_id
            FROM vm_inventory vm
            JOIN vcenters vc ON vm.vcenter_id = vc.vcenter_id
            JOIN vc_runs r ON vm.run_id = r.run_id
            WHERE vm.run_id = (SELECT MIN(v2.run_id) FROM vm_inventory v2 WHERE v2.vm_name = vm.vm_name)
              AND r.run_timestamp >= %s
            UNION ALL
            SELECT vm_name, 'Hyper-V' as platform, cluster_name as vcenter_name,
                   cpu, ram_mb, os, r.run_timestamp, r.run_id
            FROM hyperv_vms hv
            JOIN hyperv_runs r ON hv.run_id = r.run_id
            WHERE hv.run_id = (SELECT MIN(h2.run_id) FROM hyperv_vms h2 WHERE h2.vm_name = hv.vm_name)
              AND r.run_timestamp >= %s
            ORDER BY created_at DESC
        """, (cutoff_date, cutoff_date))

        for row in cursor.fetchall():
            results.append({
                'name': row['vm_name'],
                'platform': row['platform'],
                'vcenter': row['vcenter_name'],
                'cpu': row['cpu'],
                'ram_gb': round((row['ram_mb'] or 0) / 1024, 1),
                'os': row['os'],
                'created_at': str(row['created_at'])[:16],
                'run_id': row['run_id']
            })

    elif report_type == 'hosts':
        cursor.execute("""
            SELECT h.host_name, 'VMware' as platform, vc.vcenter_name,
                   h.cpu_cores, h.memory_total_gb, r.run_timestamp as created_at, r.run_id
            FROM hosts h
            JOIN vcenters vc ON h.vcenter_id = vc.vcenter_id
            JOIN vc_runs r ON h.run_id = r.run_id
            WHERE h.run_id = (SELECT MIN(h2.run_id) FROM hosts h2 WHERE h2.host_name = h.host_name)
              AND r.run_timestamp >= %s
            UNION ALL
            SELECT host_name, 'Hyper-V' as platform, cluster_name as vcenter_name,
                   cpu_cores, ROUND(memory_total_mb/1024,1) as memory_total_gb,
                   r.run_timestamp, r.run_id
            FROM hyperv_hosts h
            JOIN hyperv_runs r ON h.run_id = r.run_id
            WHERE h.run_id = (SELECT MIN(h2.run_id) FROM hyperv_hosts h2 WHERE h2.host_name = h.host_name)
              AND r.run_timestamp >= %s
            ORDER BY created_at DESC
        """, (cutoff_date, cutoff_date))

        for row in cursor.fetchall():
            results.append({
                'name': row['host_name'],
                'platform': row['platform'],
                'vcenter': row['vcenter_name'],
                'cpu_cores': row['cpu_cores'],
                'memory_gb': row['memory_total_gb'],
                'created_at': str(row['created_at'])[:16],
                'run_id': row['run_id']
            })

    elif report_type == 'datastores':
        cursor.execute("""
            SELECT d.datastore_name, 'VMware' as platform, vc.vcenter_name,
                   ROUND(d.capacity_mb/1024,1) as capacity_gb,
                   ROUND(d.free_mb/1024,1) as free_gb,
                   r.run_timestamp as created_at, r.run_id
            FROM datastores d
            JOIN vcenters vc ON d.vcenter_id = vc.vcenter_id
            JOIN vc_runs r ON d.run_id = r.run_id
            WHERE d.run_id = (SELECT MIN(d2.run_id) FROM datastores d2 WHERE d2.datastore_name = d.datastore_name)
              AND r.run_timestamp >= %s
            UNION ALL
            SELECT datastore_name, 'Hyper-V' as platform, cluster_name as vcenter_name,
                   capacity_gb, free_gb, r.run_timestamp, r.run_id
            FROM hyperv_datastores d
            JOIN hyperv_runs r ON d.run_id = r.run_id
            WHERE d.run_id = (SELECT MIN(d2.run_id) FROM hyperv_datastores d2 WHERE d2.datastore_name = d.datastore_name)
              AND r.run_timestamp >= %s
            ORDER BY created_at DESC
        """, (cutoff_date, cutoff_date))

        for row in cursor.fetchall():
            results.append({
                'name': row['datastore_name'],
                'platform': row['platform'],
                'vcenter': row['vcenter_name'],
                'capacity_gb': row['capacity_gb'],
                'free_gb': row['free_gb'],
                'created_at': str(row['created_at'])[:16],
                'run_id': row['run_id']
            })

    cursor.close()
    conn.close()
    return jsonify(results)
