async function runInventoryScript() {
    const btn = document.getElementById('refresh-btn');
    const panel = document.getElementById('script-panel');
    const output = document.getElementById('scriptOutput');
    const status = document.getElementById('script-status');

    panel.style.display = 'block';
    panel.scrollIntoView({ behavior:'smooth', block:'start' });
    btn.disabled = true; btn.classList.add('running');
    btn.innerHTML = '<span class="spinner-border spinner-border-sm me-1"></span>Running…';
    output.innerHTML = '<span style="color:#58a6ff;">▶ Starting inventory collection…</span>\n';
    status.style.display = 'block';
    status.innerHTML = `<div class="d-flex align-items-center gap-2 text-muted">
        <div class="spinner-border spinner-border-sm text-primary"></div>
        <span>Running <code>get_inventory.py</code> — may take up to 5 minutes…</span></div>`;

    const t0 = Date.now();
    function esc(s){ return (s||'').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;'); }

    try {
        const res = await fetch('/api/run_script', { method:'POST' });
        const data = await res.json();
        const elapsed = ((Date.now()-t0)/1000).toFixed(1);

        if (data.status === 'success') {
            let out = esc(data.stdout)
                .replace(/(ok=\d+)/g,'<span class="ok">$1</span>')
                .replace(/(ERROR|CRITICAL)/g,'<span class="err">$1</span>')
                .replace(/(WARNING)/g,'<span class="warn">$1</span>')
                .replace(/(INFO|Running)/g,'<span class="info">$1</span>');

            output.innerHTML = `<span class="info">▶ Finished in ${elapsed}s (exit: ${data.returncode})</span>\n\n`
                + (out || '<span style="color:#484f58;">(no output)</span>')
                + (data.stderr ? '\n<span class="warn">─── STDERR ───</span>\n'+esc(data.stderr) : '');

            status.innerHTML = data.returncode===0
                ? `<div class="d-flex align-items-center gap-2 text-success">
                     <i class="bi bi-check-circle-fill"></i>
                     <span>Updated in <strong>${elapsed}s</strong>.</span>
                     <button class="btn btn-sm btn-outline-success ms-auto" onclick="location.reload()">
                       <i class="bi bi-arrow-clockwise me-1"></i>Reload Dashboard
                     </button></div>`
                : `<div class="text-warning small"><i class="bi bi-exclamation-triangle me-1"></i>Exit code ${data.returncode}</div>`;
        } else {
            output.innerHTML = `<span class="err">✕ ${esc(data.error||'Unknown error')}</span>`;
            status.innerHTML = `<div class="text-danger small"><i class="bi bi-x-circle me-1"></i>Script failed.</div>`;
        }
    } catch(err) {
        output.innerHTML = `<span class="err">✕ Request failed: ${esc(err.message)}</span>`;
        status.innerHTML = `<div class="text-danger small"><i class="bi bi-x-circle me-1"></i>Could not reach server.</div>`;
    } finally {
        btn.disabled=false; btn.classList.remove('running');
        btn.innerHTML='<i class="bi bi-arrow-clockwise me-1"></i>Refresh Inventory';
        output.scrollTop=output.scrollHeight;
    }
}
</script>

    const DR_SITE_VCENTERS = ["MXDR", "CCDR", "UCSXDR-VC"];
    const MAIN_SITE_HYPERV = ["TE-HXCLOUD", "HVUCSX22", "HV-UCSX"];
    const DR_SITE_HYPERV = ["HVMXDR-CLUS", "HVDR-EXCHCLUS"];

    let allData = [];
    let activeCharts = {};
    let selectedResources = new Set();   // selection for navigation – does NOT filter the dashboard
    let currentPlatformFilter = "all";

    // Helper functions
    function toNum(v) {
        if (v === undefined || v === null) return 0;
        return parseFloat(v) || 0;
    }

    function formatTB(gb) {
        let gbNum = toNum(gb);
        return (gbNum / 1024).toFixed(1) + " TB";
    }

    function formatGB(gb) {
        return toNum(gb).toFixed(1) + " GB";
    }

    function formatGBorTB(gb) {
        let gbNum = toNum(gb);
        return gbNum > 1024 ? formatTB(gbNum) : formatGB(gbNum);
    }

    function getColor(percent) {
        let p = toNum(percent);
        return p > 90 ? "#dc2626" : (p > 70 ? "#f97316" : "#10b981");
    }

    // ----- Platform filter (All / VMware / Hyper-V) – filters the dashboard -----
    function setPlatformFilter(platform) {
        currentPlatformFilter = platform;
        document.querySelectorAll('.platform-filter-btn').forEach(btn => {
            if (btn.dataset.platform === platform) {
                btn.classList.add('active');
            } else {
                btn.classList.remove('active');
            }
        });
        refreshDisplay();
    }

    function getPlatformMatch(filter) {
        if (filter === 'vmware') return 'vmware';
        if (filter === 'hyperv') return 'hyper-v';
        return filter;
    }

    // Filter data by platform only – selection does NOT affect the dashboard
    function getFilteredData(data) {
        let filtered = data;
        if (currentPlatformFilter !== 'all') {
            const compareValue = getPlatformMatch(currentPlatformFilter);
            filtered = filtered.filter(row => row.platform.toLowerCase() === compareValue);
        }
        return filtered;
    }

    // ----- Navigation: detail pages -----
    // allResources = true → pass ALL resource names (navbar buttons)
    // allResources = false → pass only selected resources (dashboard buttons)
    function navigateToPage(page, allResources = true) {
        let resources = [];
        if (allResources) {
            allData.forEach(row => {
                if (row.resource_name && !resources.includes(row.resource_name)) {
                    resources.push(row.resource_name);
                }
            });
        } else {
            if (selectedResources.size === 0) {
                alert("Please select at least one resource using the checkboxes.");
                return;
            }
            resources = Array.from(selectedResources);
        }
        const selectedVals = resources.join(',');
        window.location.href = `/${page}?vcenter=${encodeURIComponent(selectedVals)}`;
    }

    // ----- Checkbox selection (for navigation only) – NO dashboard re‑render -----
    function updateSelectionInfo() {
        const infoDiv = document.getElementById('selectionInfo');
        const countSpan = document.getElementById('selectedCount');
        if (selectedResources.size > 0) {
            infoDiv.style.display = 'flex';
            countSpan.textContent = selectedResources.size;
        } else {
            infoDiv.style.display = 'none';
        }
    }

    function clearAllSelections() {
        selectedResources.clear();
        document.querySelectorAll('.resource-check').forEach(cb => {
            if (cb.disabled) return;
            cb.checked = false;
        });
        updateSelectionInfo();
        // Do NOT re-render the dashboard – just update the info bar
    }

    function attachCheckboxListeners() {
        document.querySelectorAll('.resource-check:not([disabled])').forEach(cb => {
            cb.removeEventListener('change', handleCheckboxChange);
            cb.addEventListener('change', handleCheckboxChange);
        });
    }

    // **** This function only updates the Set and the info bar – no dashboard refresh ****
    function handleCheckboxChange(e) {
        if (e.target.checked) {
            selectedResources.add(e.target.value);
        } else {
            selectedResources.delete(e.target.value);
        }
        updateSelectionInfo();
        // Dashboard is NOT re‑rendered here – this ensures multiple selections work flawlessly
    }

    // ----- CSV Export – exports only resources that belong to MAIN or DR site based on predefined lists -----
    function exportTableToCSV() {
        // Start with data filtered by platform (All/VMware/Hyper-V)
        let platformFiltered = getFilteredData(allData);
        // Further filter to only include resources that are in one of the site arrays
        let exportData = platformFiltered.filter(row => {
            const name = row.resource_name;
            return MAIN_SITE_VCENTERS.includes(name) ||
                   MAIN_SITE_HYPERV.includes(name) ||
                   DR_SITE_VCENTERS.includes(name) ||
                   DR_SITE_HYPERV.includes(name);
        });
        if (exportData.length === 0) {
            alert("No data to export for the selected platform and site filters");
            return;
        }

        let csv = [["Platform","Site","Resource Name","IP","Clusters","Hosts","Datastores","VMs","Total CPU","Used CPU","Free CPU","Total Memory","Used Memory","Free Memory","Total Storage","Used Storage","Free Storage","CPU %","Memory %","Storage %","Sub CPU Assign to VMs","Sub Memory Assign to VMs","Sub Storage Assign to VMs"]];

        exportData.forEach(row => {
            const name = row.resource_name;
            let site = "";
            if (MAIN_SITE_VCENTERS.includes(name) || MAIN_SITE_HYPERV.includes(name)) {
                site = "MAIN SITE";
            } else if (DR_SITE_VCENTERS.includes(name) || DR_SITE_HYPERV.includes(name)) {
                site = "DR SITE";
            } else {
                // Should not happen due to filter, but skip
                return;
            }
            csv.push([
                row.platform || "Unknown",
                site,
                row.resource_name || "Unknown",
                row.ip_vcenter || "-",
                row.num_clusters || 0,
                row.num_hosts || 0,
                row.num_datastores || 0,
                row.num_vms || 0,
                row.total_cores || 0,
                row.used_cores || 0,
                row.free_cores || 0,
                row.total_memory_gb/1024,
                row.used_memory_gb/1024,
                row.free_memory_gb/1024,
                row.total_storage_gb/1024,
                row.used_storage_gb/1024,
                row.free_storage_gb/1024,
                row.cpu_usage_percent?.toFixed(1) + "%" || "0%",
                row.memory_usage_percent?.toFixed(1) + "%" || "0%",
                row.storage_usage_percent?.toFixed(1) + "%" || "0%",
                row.subscription_cpu_ratio || 'N/A',
                row.subscription_memory_ratio || 'N/A',
                row.subscription_storage_ratio || 'N/A'
            ]);
        });

        let blob = new Blob([csv.map(r => r.join(",")).join("\n")], { type: "text/csv" });
        let link = document.createElement("a");
        link.href = URL.createObjectURL(blob);
        link.download = "infrastructure_export.csv";
        link.click();
        URL.revokeObjectURL(link.href);
    }

    // ----- Resource summary (top cards) – uses platform filter only -----
    function updateResourceSummary(data) {
        let filteredData = getFilteredData(data);

        let mainFreeCores = 0, mainFreeMemoryGB = 0, mainFreeStorageGB = 0, mainTotalVMs = 0;
        let drFreeCores = 0, drFreeMemoryGB = 0, drFreeStorageGB = 0, drTotalVMs = 0;

        let mainResources = [];
        let drResources = [];

        filteredData.forEach(row => {
            const resourceName = row.resource_name;
            const freeCores = toNum(row.free_cores);
            const freeMemory = toNum(row.free_memory_gb);
            const freeStorage = toNum(row.free_storage_gb);
            const totalVMs = toNum(row.num_vms);

            if (row.site_name === 'B7') {
                mainResources.push({name: resourceName, platform: row.platform});
                mainFreeCores += freeCores;
                mainFreeMemoryGB += freeMemory;
                mainFreeStorageGB += freeStorage;
                mainTotalVMs += totalVMs;
            } else if (row.site_name === 'DR') {
                drResources.push({name: resourceName, platform: row.platform});
                drFreeCores += freeCores;
                drFreeMemoryGB += freeMemory;
                drFreeStorageGB += freeStorage;
                drTotalVMs += totalVMs;
            }
        });

        document.getElementById("mainFreeCores").innerHTML = mainFreeCores.toLocaleString();
        document.getElementById("mainFreeMemory").innerHTML = formatTB(mainFreeMemoryGB);
        document.getElementById("mainFreeStorage").innerHTML = formatTB(mainFreeStorageGB);
        document.getElementById("mainTotalVMs").innerHTML = mainTotalVMs.toLocaleString();

        let mainTags = MAIN_SITE_VCENTERS.map(vc => {
            let found = mainResources.find(r => r.name === vc);
            let statusClass = found ? 'online' : 'offline';
            let icon = found ? 'fa-check-circle' : 'fa-circle-exclamation';
            return `<span class="vcenter-tag ${statusClass}"><i class="fas ${icon}"></i> ${vc}</span>`;
        }).join('');
        mainTags += MAIN_SITE_HYPERV.map(hv => {
            let found = mainResources.find(r => r.name === hv);
            let statusClass = found ? 'online' : 'offline';
            return `<span class="vcenter-tag hyperv ${statusClass}"><i class="fab fa-windows"></i> ${hv}</span>`;
        }).join('');
        document.getElementById("mainVcenterTags").innerHTML = mainTags;

        document.getElementById("drFreeCores").innerHTML = drFreeCores.toLocaleString();
        document.getElementById("drFreeMemory").innerHTML = formatTB(drFreeMemoryGB);
        document.getElementById("drFreeStorage").innerHTML = formatTB(drFreeStorageGB);
        document.getElementById("drTotalVMs").innerHTML = drTotalVMs.toLocaleString();

        let drTags = DR_SITE_VCENTERS.map(vc => {
            let found = drResources.find(r => r.name === vc);
            let statusClass = found ? 'online' : 'offline';
            let icon = found ? 'fa-check-circle' : 'fa-circle-exclamation';
            return `<span class="vcenter-tag ${statusClass}"><i class="fas ${icon}"></i> ${vc}</span>`;
        }).join('');
        drTags += DR_SITE_HYPERV.map(hv => {
            let found = drResources.find(r => r.name === hv);
            let statusClass = found ? 'online' : 'offline';
            return `<span class="vcenter-tag hyperv ${statusClass}"><i class="fab fa-windows"></i> ${hv}</span>`;
        }).join('');
        document.getElementById("drVcenterTags").innerHTML = drTags;
    }

    // ----- Site chart rendering -----
    function renderSiteChart(siteId, avgCpu, avgMem, avgStorage) {
        const canvasId = `chart-${siteId.replace(/\s/g, '')}`;
        const canvas = document.getElementById(canvasId);
        if (!canvas) return;
        const ctx = canvas.getContext('2d');
        if (!ctx) return;
        if (activeCharts[canvasId]) activeCharts[canvasId].destroy();
        activeCharts[canvasId] = new Chart(ctx, {
            type: 'doughnut',
            data: {
                labels: ['CPU Usage %', 'Memory Usage %', 'Storage Usage %'],
                datasets: [{
                    data: [avgCpu, avgMem, avgStorage],
                    backgroundColor: ['#3b82f6', '#8b5cf6', '#10b981'],
                    borderWidth: 0,
                    borderRadius: 8,
                    cutout: '65%'
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: true,
                plugins: {
                    legend: { position: 'bottom', labels: { font: { size: 8 }, boxWidth: 8 } },
                    tooltip: { callbacks: { label: (ctx) => `${ctx.label}: ${ctx.raw.toFixed(1)}%` } }
                }
            }
        });
    }

    function getDisplaySite(siteName) {
        switch(siteName) {
            case 'B7': return 'B7 Cloud';
            case 'DR': return 'DR Cloud';
            case 'B90': return 'B90 Cloud';
            case 'TESTDEV': return 'Test & Dev';
            default: return siteName;
        }
    }

    // ----- Load run options (called once on page load) -----
    function loadRuns() {
        return fetch("/api/unified/runs")
            .then(res => res.json())
            .then(data => {
                const vmwareSelector = document.getElementById("vmwareRunFilter");
                const hypervSelector = document.getElementById("hypervRunFilter");
                if (data.runs && Array.isArray(data.runs)) {
                    let vmwareRuns = data.runs.filter(r => r.platform === 'VMware');
                    let hypervRuns = data.runs.filter(r => r.platform === 'Hyper-V');

                    vmwareSelector.innerHTML = '<option value="">Latest VMware Run</option>' +
                        vmwareRuns.map(r => `<option value="${r.run_id}">Run ${r.run_id} - ${r.run_timestamp || 'N/A'}</option>`).join('');
                    hypervSelector.innerHTML = '<option value="">Latest Hyper-V Run</option>' +
                        hypervRuns.map(r => `<option value="${r.run_id}">Run ${r.run_id} - ${r.run_timestamp || 'N/A'}</option>`).join('');

                    vmwareSelector.onchange = () => loadUnifiedSiteData();
                    hypervSelector.onchange = () => loadUnifiedSiteData();
                }
            })
            .catch(err => console.error("Failed to load runs:", err));
    }

    // ----- Load data from backend (all four sites) using selected run IDs -----
    function loadUnifiedSiteData() {
        let selectedVmwareRun = document.getElementById("vmwareRunFilter")?.value;
        let selectedHypervRun = document.getElementById("hypervRunFilter")?.value;

        document.getElementById("powerbiGrid").innerHTML = '<div class="loader-center text-center py-5"><i class="fas fa-spinner fa-pulse fa-2x"></i><p>Loading infrastructure data...</p></div>';
        document.getElementById("groupedTablesContainer").innerHTML = '<div class="text-center py-4"><i class="fas fa-spinner fa-pulse"></i> Loading tables...</div>';

        const sites = ['b7', 'dr', 'b90', 'testdev'];
        const promises = sites.map(site => {
            let url = `/api/unified/site/${site}`;
            let params = [];
            if (selectedVmwareRun && selectedVmwareRun !== "") params.push(`run_id_vmware=${selectedVmwareRun}`);
            if (selectedHypervRun && selectedHypervRun !== "") params.push(`run_id_hyperv=${selectedHypervRun}`);

        });

        Promise.all(promises)
            .then(results => {

                results.forEach(result => {
                    if (result.data && Array.isArray(result.data)) {
                        combinedData = combinedData.concat(result.data);
                    }
                });
                allData = combinedData;
                console.log("Loaded all data:", allData);

                updateResourceSummary(allData);
                buildPowerBICards(allData);
                renderGroupedTables(allData);
            })
            .catch(err => {
                console.error("Error loading data:", err);
                document.getElementById("powerbiGrid").innerHTML = '<div class="text-center py-4 text-danger">Failed to load data: ' + err.message + '</div>';
                document.getElementById("groupedTablesContainer").innerHTML = '<div class="text-center py-4 text-danger">Failed to load data</div>';
            });
